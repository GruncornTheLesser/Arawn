Arawn::Graph::~Graph() noexcept { 
	// iterate through passes


	for (uint32_t p = 0; p < pass_count[Queue::GRAPHICS]; ++p) {
		auto& pass = graphics_passes[p];
		
		vkDestroyRenderPass(engine.device, pass.renderpass, nullptr);

		for (uint32_t s = 0; s < pass.subpass_count; ++s) {
			auto& subpass = pass.subpasses[s];
			std::destroy_at(&subpass.record);
			vkDestroyPipeline(engine.device, subpass.pipeline, nullptr);
			vkDestroyPipelineLayout(engine.device, subpass.layout, nullptr);
		}
		
		for (uint32_t f = 0; f < frame_count; ++f) {
			auto& frame = pass.frames[f];
			vkDestroyFramebuffer(engine.device, frame.framebuffer, nullptr);
		}
	}

	for (uint32_t p = 0; p < pass_count[Queue::COMPUTE]; ++p) {
		auto& pass = compute_passes[p];

		std::destroy_at(&pass.record);
		vkDestroyPipeline(engine.device, pass.pipeline, nullptr);
		vkDestroyPipelineLayout(engine.device, pass.layout, nullptr);

		for (uint32_t f = 0; f < frame_count; ++f) {
			auto& frame = pass.frames[f];
			
		}
	}
	
	
	allocator.release();
}
